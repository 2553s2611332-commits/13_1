
st.subheader("➡️ 다음 단계로 이동")

col1, col2 = st.columns(2)

with col1:
    if st.button("⬅️ 이전 단계"):
        if st.session_state.step_index > 0:
            st.session_state.step_index -= 1
            st.session_state.step_selector = step_names[
                st.session_state.step_index
            ]
            st.rerun()
        else:
            st.info("첫 번째 단계입니다.")

with col2:
    if st.button("다음 단계 ➡️"):
        if st.session_state.step_index < len(step_names) - 1:
            st.session_state.step_index += 1
            st.session_state.step_selector = step_names[
                st.session_state.step_index
            ]
            st.rerun()
        else:
            st.success(
                "마지막 단계입니다. 주문이 완료되었는지 확인하세요."
            )

if st.button("🏠 처음부터 다시 시작"):
    st.session_state.step_index = 0
    st.session_state.step_selector = step_names[0]
    st.rerun()
